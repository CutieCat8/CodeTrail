assert.deepEqual(errors,[]);fs.writeFileSync('/tmp/sea-quest-browser/java-report.json',JSON.stringify(report,null,2));console.log(JSON.stringify({...report,routes:report.routes.length},null,2));await context.close();await browser.close();})().catch(async e=>{console.error(e);await browser?.close();process.exitCode=1;});
